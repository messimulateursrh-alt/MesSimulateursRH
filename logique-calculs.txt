function calculerIndemniteKilometrique(puissanceCV, distanceKm, electrique){
// Barème forfaitaire par tranche de distance annuelle, cf. BAREME_KM_2026 ci-dessus.

if (electrique){
montant = montant * 1.20;
}
return {

montantAvantMajoration: montantAvantMajoration,

};
}
var BAREME_IR_2026 = [
// Barème de l'impôt sur le revenu applicable aux revenus 2025 (déclaration 2026), revalorisé de +0,9%.
// Vérifié sur 2 sources indépendantes (service-public.gouv.fr F1419, legifiscal.fr) : seuils concordants.

{seuil: Infinity, taux: 0.45}
];
function calculerPartsFiscales(situation, nbEnfants, gardeAlternee, parentIsole, nbInvalides){
// Art. 194 et 195 du CGI. situation: 'seul' ou 'couple'. nbEnfants: entier >= 0.
// Parent isolé (case T, art. 195-1 b) : ne s'applique qu'à situation === 'seul', bonus de 0,5 part
// (0,25 en garde alternée) réservé au 1er enfant, en plus de la majoration normale de ce même enfant
// (vérifié sur service-public.gouv.fr F35120 : 1 personne seule + 1 enfant en case T = 2 parts, pas 1,5).
var parts = situation === 'couple' ? 2 : 1;
for (var i = 1; i <= nbEnfants; i++){
var frac = i <= 2 ? 0.5 : 1.0;
if (gardeAlternee) frac = frac / 2;
parts += frac;
if (i === 1 && parentIsole && situation === 'seul'){
parts += gardeAlternee ? 0.25 : 0.5;
}
}
parts += 0.5 * (nbInvalides || 0);
return parts;
}
function trancheMarginale(quotientFamilial){
// Renvoie la tranche du barème applicable à un quotient familial donné (revenu net imposable / nombre de parts).
for (var i = 0; i < BAREME_IR_2026.length; i++){
if (quotientFamilial <= BAREME_IR_2026[i].seuil){
var seuilBas = i === 0 ? 0 : BAREME_IR_2026[i-1].seuil;
return {
taux: BAREME_IR_2026[i].taux,
seuilBas: seuilBas,
seuilHaut: BAREME_IR_2026[i].seuil
};
}
}
return BAREME_IR_2026[BAREME_IR_2026.length - 1];
}
// --- Comparateur mariage/PACS, ajouté le 14/09/2026 ---
// Paramètres fiscaux 2026 (revenus 2025), vérifiés sur Légifrance (art. 197 CGI dans sa version issue de la loi
// n°2026-103 du 19 février 2026) et le BOFiP (BOI-IR-LIQ-20-20-20 et BOI-IR-LIQ-20-20-30, publiés le 07/04/2026).
var PLAFOND_DEMI_PART_QF_2026 = 1807;   // plafond de droit commun, par demi-part supplémentaire
var DECOTE_FORFAIT_SEUL_2026 = 897;     // décote, montant forfaitaire personne seule
var DECOTE_FORFAIT_COUPLE_2026 = 1483;  // décote, montant forfaitaire couple soumis à imposition commune
var DECOTE_TAUX_2026 = 0.4525;          // taux appliqué à l'impôt brut dans la formule de décote
function impotBaremeParts(revenuImposable, parts){
// Impôt brut issu du seul barème progressif : on applique le barème au quotient (revenu / parts), puis on remultiplie.
if (!revenuImposable || revenuImposable <= 0 || !parts || parts <= 0) return 0;
var quotient = revenuImposable / parts;
var impotParPart = 0;
var seuilBas = 0;
for (var i = 0; i < BAREME_IR_2026.length; i++){
var seuilHaut = BAREME_IR_2026[i].seuil;
if (quotient > seuilBas){
impotParPart += (Math.min(quotient, seuilHaut) - seuilBas) * BAREME_IR_2026[i].taux;
}
seuilBas = seuilHaut;
}
return impotParPart * parts;
}
function impotSurRevenuEstime(revenuImposable, parts, situation){
// Enchaînement officiel : barème → plafonnement de l'avantage des demi-parts supplémentaires → décote.
// Le plafonnement ne porte JAMAIS sur le quotient conjugal : les 2 parts d'un couple marié ou pacsé ne sont pas
// plafonnées (BOI-IR-LIQ-20-20-20). Seules les demi-parts supplémentaires (enfants, etc.) le sont.
// Périmètre volontairement limité : ni réductions/crédits d'impôt, ni contribution exceptionnelle sur les hauts
// revenus, ni pension alimentaire, ni système du quotient sur revenus exceptionnels.
var partsBase = situation === 'couple' ? 2 : 1;
var impotBrut = impotBaremeParts(revenuImposable, parts);
var impotSansDemiParts = impotBaremeParts(revenuImposable, partsBase);
var demiPartsSupp = Math.max(0, (parts - partsBase) * 2);
var avantage = impotSansDemiParts - impotBrut;
var plafond = demiPartsSupp * PLAFOND_DEMI_PART_QF_2026;
var plafonne = false;
var impot = impotBrut;
if (demiPartsSupp > 0 && avantage > plafond){
impot = impotSansDemiParts - plafond;
plafonne = true;
}
var forfaitDecote = situation === 'couple' ? DECOTE_FORFAIT_COUPLE_2026 : DECOTE_FORFAIT_SEUL_2026;
var decote = Math.max(0, forfaitDecote - DECOTE_TAUX_2026 * impot);
decote = Math.min(decote, impot);
return {
impot: Math.max(0, impot - decote),
impotAvantDecote: impot,
decote: decote,
avantageQF: avantage,
plafondQF: plafond,
plafonne: plafonne,
parts: parts
};
}
function comparerMariagePacs(revenu1, revenu2, nbEnfants, enfantsChez){
// Compare l'impôt de deux personnes vivant ensemble SANS être mariées ni pacsées (deux foyers fiscaux distincts)
// à celui du même couple marié ou pacsé (un seul foyer, imposition commune).
// Hypothèse explicitement posée et affichée à l'utilisateur : les deux personnes vivent ensemble dans les deux cas,
// donc la demi-part de parent isolé (case T, art. 195-1 b du CGI) ne s'applique ni avant ni après — elle suppose
// de vivre seul. C'est ce qui rend la comparaison honnête : on ne compare pas un célibataire isolé à un couple.
// enfantsChez : 1 ou 2 — parent au foyer duquel les enfants sont rattachés tant que le couple n'est pas marié/pacsé
// (les enfants ne peuvent être comptés que dans un seul foyer fiscal).
var n = Math.max(0, nbEnfants || 0);
var parts1 = calculerPartsFiscales('seul', enfantsChez === 1 ? n : 0, false, false, 0);
var parts2 = calculerPartsFiscales('seul', enfantsChez === 2 ? n : 0, false, false, 0);
var r1 = impotSurRevenuEstime(revenu1, parts1, 'seul');
var r2 = impotSurRevenuEstime(revenu2, parts2, 'seul');
var partsCommun = calculerPartsFiscales('couple', n, false, false, 0);
var commun = impotSurRevenuEstime((revenu1 || 0) + (revenu2 || 0), partsCommun, 'couple');
var totalSepare = r1.impot + r2.impot;
return {
separe1: r1,
separe2: r2,
totalSepare: totalSepare,
commun: commun,
partsCommun: partsCommun,
ecart: totalSepare - commun.impot
};
}
function calculerIndemniteActivitePartielle(tauxHoraireBrutHabituel, nbHeuresChomees){
// Art. R5122-18 du Code du travail — 60% de la rémunération horaire brute de référence,
// plafonnée à 4,5×SMIC brut horaire, avec plancher au SMIC net horaire.
var remunerationRefHoraire = Math.min(tauxHoraireBrutHabituel, BAREME_2026.ACTIVITE_PARTIELLE_PLAFOND_SMIC_MULT * BAREME_2026.SMIC_BRUT_HORAIRE);
var indemniteHoraireBrute = BAREME_2026.ACTIVITE_PARTIELLE_TAUX * remunerationRefHoraire;
indemniteHoraireBrute = Math.max(indemniteHoraireBrute, BAREME_2026.SMIC_NET_HORAIRE);
var indemniteBrutTotale = indemniteHoraireBrute * nbHeuresChomees;

var baseCSGCRDS = indemniteBrutTotale * (1 - BAREME_2026.ACTIVITE_PARTIELLE_ABATTEMENT_FRAIS);
var montantCSGCRDS = baseCSGCRDS * BAREME_2026.ACTIVITE_PARTIELLE_CSG_CRDS_TAUX;

// Écrêtement : le net ne doit jamais descendre sous le SMIC brut horaire × heures chômées.
var plancherPostPrelevement = BAREME_2026.SMIC_BRUT_HORAIRE * nbHeuresChomees;
var indemniteNette = indemniteBrutTotale - montantCSGCRDS;
if (indemniteNette < plancherPostPrelevement){
montantCSGCRDS = Math.max(0, indemniteBrutTotale - plancherPostPrelevement);
indemniteNette = indemniteBrutTotale - montantCSGCRDS;
}

return {
indemniteHoraireBrute: indemniteHoraireBrute,
indemniteBrutTotale: indemniteBrutTotale,
csgCrds: montantCSGCRDS,
indemniteNette: indemniteNette
};
}
function estimerCongesPayes(remunerationRef, salaireMensuel, joursAIndemniser){

};
}
function estimerNetDepuisBrut(brutMensuel, statutCadre){
var coef = statutCadre ? BAREME_2026.COEF_NET_CADRE : BAREME_2026.COEF_NET_NON_CADRE;
var net = brutMensuel * coef;
var netImposable = net * 1.0169; // réintégration approximative de la CSG/CRDS non déductible (~1.7% du brut)
return { net: net, netImposable: netImposable, coef: coef };
}
function estimerBrutDepuisNet(netMensuel, statutCadre){
var coef = statutCadre ? BAREME_2026.COEF_NET_CADRE : BAREME_2026.COEF_NET_NON_CADRE;
var brut = netMensuel / coef;
var netImposable = netMensuel * 1.0169; // réintégration approximative de la CSG/CRDS non déductible (~1.7% du net)
return { brut: brut, netImposable: netImposable, coef: coef };
}
function detailCotisationsSalariales(brutMensuel, statutCadre){
var brutPlafonne = Math.min(brutMensuel, BAREME_2026.PASS_MENSUEL);
var assietteCSG = brutMensuel * BAREME_2026.ABATTEMENT_ASSIETTE_CSG_CRDS;
var vieillessePlafonnee = brutPlafonne * BAREME_2026.TAUX_VIEILLESSE_PLAFONNEE;
var vieillesseDeplafonnee = brutMensuel * BAREME_2026.TAUX_VIEILLESSE_DEPLAFONNEE;
var agircArrco = brutPlafonne * BAREME_2026.TAUX_AGIRC_ARRCO_T1_SALARIE;
var ceg = brutPlafonne * BAREME_2026.TAUX_CEG_T1_SALARIE;
var apec = statutCadre ? Math.min(brutMensuel, 4 * BAREME_2026.PASS_MENSUEL) * BAREME_2026.TAUX_APEC_SALARIE : 0;
var csgDeductible = assietteCSG * BAREME_2026.TAUX_CSG_DEDUCTIBLE;
var csgNonDeductible = assietteCSG * BAREME_2026.TAUX_CSG_NON_DEDUCTIBLE;
var crds = assietteCSG * BAREME_2026.TAUX_CRDS;
var total = vieillessePlafonnee + vieillesseDeplafonnee + agircArrco + ceg + apec + csgDeductible + csgNonDeductible + crds;
return {
brutPlafonne: brutPlafonne,
vieillessePlafonnee: vieillessePlafonnee,
vieillesseDeplafonnee: vieillesseDeplafonnee,
agircArrco: agircArrco,
ceg: ceg,
apec: apec,
csgDeductible: csgDeductible,
csgNonDeductible: csgNonDeductible,
crds: crds,
total: total,
tauxTotal: total / brutMensuel
};
}
// Statuts étendus (fonction publique, indépendant BNC, portage salarial) : coefficient global ou chaîne de calcul dédiée,
// distincts de estimerNetDepuisBrut/estimerBrutDepuisNet (non-cadre/cadre) pour ne pas modifier leur signature — utilisées
// ailleurs sur le site (passage-temps-partiel.html, augmentation-salaire.html) avec le paramètre booléen d'origine.
// statut : 'fonction-publique' | 'independant-ssi' | 'independant-cipav' | 'portage'.
function estimerNetStatutEtendu(montantAvantCotisations, statut){
if (statut === 'fonction-publique'){
var net = montantAvantCotisations * BAREME_2026.COEF_NET_FONCTION_PUBLIQUE;
return { net: net, coef: BAREME_2026.COEF_NET_FONCTION_PUBLIQUE };
}
if (statut === 'independant-ssi' || statut === 'independant-cipav'){
var coefInd = statut === 'independant-ssi' ? BAREME_2026.COEF_NET_INDEPENDANT_SSI : BAREME_2026.COEF_NET_INDEPENDANT_CIPAV;
var netInd = montantAvantCotisations * coefInd;
return { net: netInd, coef: coefInd };
}
if (statut === 'portage'){
// CA facturé HT -> frais de gestion de la société de portage -> brut équivalent (charges patronales moyennes) -> net (coefficient cadre).
var apresGestion = montantAvantCotisations * (1 - BAREME_2026.PORTAGE_FRAIS_GESTION);
var brutEquivalent = apresGestion / (1 + BAREME_2026.CHARGES_PATRONALES_MOYENNES);
var r = estimerNetDepuisBrut(brutEquivalent, true); // assimilé cadre : profil le plus fréquent en portage salarial
return { net: r.net, coef: r.coef, apresGestion: apresGestion, brutEquivalent: brutEquivalent };
}
return null;
}
function estimerMontantAvantCotisationsDepuisNet(net, statut){
if (statut === 'fonction-publique'){
return { montant: net / BAREME_2026.COEF_NET_FONCTION_PUBLIQUE, coef: BAREME_2026.COEF_NET_FONCTION_PUBLIQUE };
}
if (statut === 'independant-ssi' || statut === 'independant-cipav'){
var coefInd = statut === 'independant-ssi' ? BAREME_2026.COEF_NET_INDEPENDANT_SSI : BAREME_2026.COEF_NET_INDEPENDANT_CIPAV;
return { montant: net / coefInd, coef: coefInd };
}
if (statut === 'portage'){
var rInv = estimerBrutDepuisNet(net, true);
var brutEquivalent = rInv.brut;
var apresGestion = brutEquivalent * (1 + BAREME_2026.CHARGES_PATRONALES_MOYENNES);
var ca = apresGestion / (1 - BAREME_2026.PORTAGE_FRAIS_GESTION);
return { montant: ca, coef: rInv.coef, apresGestion: apresGestion, brutEquivalent: brutEquivalent };
}
return null;
}
function estimerARE(salaireMensuelBrut){
var sjr = (salaireMensuelBrut * 12) / 365;
var brancheA = 0.404 * sjr + BAREME_2026.ARE_PARTIE_FIXE_JOUR;
var brancheB = 0.57 * sjr;
var brut = Math.max(brancheA, brancheB);
var plafond = BAREME_2026.ARE_PLAFOND_PART_SJR * sjr;
brut = Math.min(brut, plafond);
brut = Math.max(brut, BAREME_2026.ARE_PLANCHER_JOUR);
return { sjr: sjr, jour: brut, mois: brut * 30.42 };
}
function estimerCumulARE(salaireAncien, nouveauSalaire){
var are = estimerARE(salaireAncien);
var resteAllocation = are.mois - (0.70 * nouveauSalaire);
var joursBruts = 0;
var montantMensuel = 0;
if (resteAllocation > 0 && are.jour > 0){
joursBruts = Math.round(resteAllocation / are.jour);
joursBruts = Math.min(Math.max(joursBruts, 0), 30);
montantMensuel = joursBruts * are.jour;
}
var joursIndemnisables = joursBruts;
var cumulTotal = nouveauSalaire + montantMensuel;
var plafond = salaireAncien;
var plafondDepasse = false;
if (cumulTotal > plafond){
plafondDepasse = true;
montantMensuel = Math.max(plafond - nouveauSalaire, 0);
joursIndemnisables = are.jour > 0 ? Math.round(montantMensuel / are.jour) : 0;
cumulTotal = nouveauSalaire + montantMensuel;
}
return { are: are, resteAllocation: resteAllocation, joursIndemnisables: joursIndemnisables, montantMensuel: montantMensuel, cumulTotal: cumulTotal, plafond: plafond, plafondDepasse: plafondDepasse, aucunCumulPossible: resteAllocation <= 0 };
}
function indemniteLegale(salaireRef, ancienneteAnnees){
if (ancienneteAnnees <= 10){
return ancienneteAnnees * (salaireRef / 4);
}
var part1 = 10 * (salaireRef / 4);
var part2 = (ancienneteAnnees - 10) * (salaireRef / 3);
return part1 + part2;
}
function anneesDecimal(annees, mois){
return (parseInt(annees || 0, 10)) + (parseInt(mois || 0, 10)) / 12;
}
function indemniteSyntecCadre(salaireRef, ancienneteAnnees){
// Convention Syntec (bureaux d'études techniques, IDCC 1486) — statut cadre.
// Formule : 1/3 de mois de salaire par année d'ancienneté dès 2 ans, sans plafond (avenant du 22 mai 2023).
// En-dessous de 2 ans d'ancienneté, la convention n'est pas plus favorable que le barème légal.
if (ancienneteAnnees < 2){
return indemniteLegale(salaireRef, ancienneteAnnees);
}
return ancienneteAnnees * (salaireRef / 3);
}
function indemniteBatimentCadre(salaireRef, ancienneteAnnees){
// Convention collective nationale des cadres du bâtiment du 1er juin 2004 (IDCC 2420).
// Formule : 3/10 de mois par année jusqu'à 10 ans, puis 6/10 de mois par année au-delà de 10 ans, plafonnée à 15 mois.
// En-dessous de 2 ans d'ancienneté, la convention n'est pas plus favorable que le barème légal.
if (ancienneteAnnees < 2){
return indemniteLegale(salaireRef, ancienneteAnnees);
}
var anneesJusqua10 = Math.min(ancienneteAnnees, 10);
var anneesAuDela10 = Math.max(0, ancienneteAnnees - 10);
var indemnite = (0.30 * anneesJusqua10 * salaireRef) + (0.60 * anneesAuDela10 * salaireRef);
var plafond = 15 * salaireRef;
return Math.min(indemnite, plafond);
}
function indemniteTransportETAM(salaireRef, ancienneteAnnees){
// Convention collective nationale des transports routiers et activités auxiliaires (IDCC 16) — techniciens et agents de maîtrise (ETAM).
// Formule (source : Légifrance art. dédié + LégiSocial, Éditions Tissot, Juritravail, Abacus-Conseil, Sidecare, concordants) :
// 3/10 de mois de salaire par année de présence dès 3 ans d'ancienneté, taux appliqué à la totalité de l'ancienneté (pas seulement à la fraction au-delà du seuil).
// En-dessous de 3 ans d'ancienneté, la convention n'est pas plus favorable que le barème légal.
if (ancienneteAnnees < 3){
return indemniteLegale(salaireRef, ancienneteAnnees);
}
return ancienneteAnnees * (salaireRef * 0.30);
}
function indemniteTransportCadre(salaireRef, ancienneteAnnees){
// Convention collective nationale des transports routiers et activités auxiliaires (IDCC 16) — ingénieurs et cadres.
// Formule (mêmes sources que ci-dessus) : 4/10 de mois de salaire par année de présence dès 3 ans d'ancienneté, taux appliqué à la totalité de l'ancienneté.
// Ne couvre pas la règle de cumul en cas de promotion interne depuis une catégorie inférieure, ni une éventuelle majoration liée à l'âge (seuil et existence non confirmés par des sources concordantes) — non intégrées par prudence.
// En-dessous de 3 ans d'ancienneté, la convention n'est pas plus favorable que le barème légal.
if (ancienneteAnnees < 3){
return indemniteLegale(salaireRef, ancienneteAnnees);
}
return ancienneteAnnees * (salaireRef * 0.40);
}
function indemniteHCR(salaireRef, ancienneteAnnees){
// Convention collective nationale des hôtels, cafés, restaurants (HCR, IDCC 1979), art. 32 — tous statuts, sans distinction cadre/non-cadre (source : Légifrance, texte primaire, confirmé par 5 sources secondaires concordantes).
// Formule : 1/10 de mois par année d'ancienneté jusqu'à 10 ans, puis 1/15 de mois par année au-delà de 10 ans.
// Cette formule est structurellement toujours moins favorable que le barème légal (1/4 puis 1/3) — affichée par transparence, la comparaison automatique retient de toute façon le barème légal.
if (ancienneteAnnees < 2){
return indemniteLegale(salaireRef, ancienneteAnnees);
}
var anneesJusqua10 = Math.min(ancienneteAnnees, 10);
var anneesAuDela10 = Math.max(0, ancienneteAnnees - 10);
return (anneesJusqua10 * (salaireRef / 10)) + (anneesAuDela10 * (salaireRef / 15));
}
function indemniteSyntecETAM(salaireRef, ancienneteAnnees){
// Convention Syntec (bureaux d'études techniques, IDCC 1486), art. 4.5 — statut non-cadre (ETAM), texte consolidé Légifrance vérifié directement.
// Formule (depuis le 1er mai 2023, avenants n°46 du 16/07/2021 et n°2 du 27/10/2022, étendus par arrêté du 5 avril 2023) :
// 1/4 de mois de salaire par année d'ancienneté jusqu'à 10 ans, puis 1/3 de mois par année au-delà — c'est-à-dire structurellement
// la même formule que le barème légal (Art. R1234-2), sans plafond. Incluse ici par transparence : la convention ne procure aucun
// avantage supplémentaire pour ce statut, contrairement au statut cadre (voir indemniteSyntecCadre).
if (ancienneteAnnees <= 10){
return ancienneteAnnees * (salaireRef / 4);
}
var part1 = 10 * (salaireRef / 4);
var part2 = (ancienneteAnnees - 10) * (salaireRef / 3);
return part1 + part2;
}
function indemniteBatimentETAM(salaireRef, ancienneteAnnees){
// Convention collective nationale des employés, techniciens et agents de maîtrise du bâtiment du 12 juillet 2006 (IDCC 2609).
// Formule (source : LégiSocial, Fiche-paie.net, Lexiconvention, Abacus-Conseil, FFB, concordantes) :
// 2,5/10 de mois par année d'ancienneté jusqu'à 15 ans, puis 3,5/10 de mois par année au-delà de 15 ans, plafonnée à 10 mois de salaire.
// Majoration de 10 % à 55 ans et plus non intégrée au calcul (âge non demandé par ce simulateur) — signalée en avertissement textuel.
// En-dessous de 2 ans d'ancienneté, la convention n'est pas plus favorable que le barème légal.
if (ancienneteAnnees < 2){
return indemniteLegale(salaireRef, ancienneteAnnees);
}
var anneesJusqua15 = Math.min(ancienneteAnnees, 15);
var anneesAuDela15 = Math.max(0, ancienneteAnnees - 15);
var indemnite = (0.25 * anneesJusqua15 * salaireRef) + (0.35 * anneesAuDela15 * salaireRef);
var plafond = 10 * salaireRef;
return Math.min(indemnite, plafond);
}
function indemniteBatimentOuvrier(salaireRef, ancienneteAnnees){
// Conventions collectives nationales des ouvriers du bâtiment du 8 octobre 1990, IDCC 1596 (jusqu'à 10 salariés) et IDCC 1597
// (plus de 10 salariés), art. 10.3 — même formule pour les deux. Source : LégiSocial (1596 et 1597), Éditions Tissot, FFB,
// Mon Indemnité, concordantes.
// Formule : 1/10 de mois par année d'ancienneté de 2 à 5 ans ; à partir de 5 ans, 3/20 de mois par année, appliqué à la totalité
// de l'ancienneté (bascule de taux rétroactive, pas un cumul de tranches) ; majoration de 1/20 de mois par année supplémentaire
// au-delà de 15 ans. Majoration de 10 % à 55 ans et plus non intégrée (âge non demandé) — signalée en avertissement textuel.
// Cette formule est presque toujours moins favorable que le barème légal (0,15/an contre 0,25/an pour les 10 premières années) —
// incluse par transparence, la comparaison automatique retient de toute façon le plus favorable des deux.
// En-dessous de 2 ans d'ancienneté, la convention n'est pas plus favorable que le barème légal.
if (ancienneteAnnees < 2){
return indemniteLegale(salaireRef, ancienneteAnnees);
}
var base;
if (ancienneteAnnees < 5){
base = ancienneteAnnees * (salaireRef * 0.10);
} else {
base = ancienneteAnnees * (salaireRef * 0.15);
}
var anneesAuDela15 = Math.max(0, ancienneteAnnees - 15);
var majorationAncienne = anneesAuDela15 * (salaireRef * 0.05);
return base + majorationAncienne;
}
function preavisConventionCollective(convention, typeRupture, ancienneteAnnees){
// Durées de préavis (hors licenciement pour faute grave/lourde, qui supprime le préavis quelle que soit la convention).
// Sources croisées : code.travail.gouv.fr (contributions dédiées par IDCC) + LégiSocial + Dairia Avocats (IDCC 16) / Juritravail (IDCC 1979), concordants sauf mention contraire ci-dessous.
// Retourne { mois: <équivalent en mois, pour le calcul de l'indemnité de dispense>, texte: <durée affichée> } ou null si la convention/le type ne sont pas couverts.
if (convention === 'transport_etam'){
// IDCC 16, techniciens et agents de maîtrise (groupes 1 à 5). Démission : 1 mois quelle que soit l'ancienneté. Licenciement : 1 mois en-dessous de 2 ans, 2 mois au-delà — comble le vide légal en-dessous de 6 mois (aucun minimum par défaut dans le Code du travail à ce stade).
if (typeRupture === 'demission'){
return { mois: 1, texte: '1 mois' };
}
return ancienneteAnnees < 2 ? { mois: 1, texte: '1 mois' } : { mois: 2, texte: '2 mois' };
}
if (convention === 'transport_cadre'){
// IDCC 16, ingénieurs et cadres. 3 mois quels que soient l'ancienneté et le type de rupture (démission ou licenciement).
return { mois: 3, texte: '3 mois' };
}
if (convention === 'hcr_employe'){
// HCR (IDCC 1979), employés. Démission : 8 jours (< 6 mois), 15 jours (6 mois à 2 ans), 1 mois (2 ans et plus). Licenciement : 8 jours (< 6 mois, seuil confirmé par une seule source — LégiSocial — à traiter avec prudence), 1 mois (6 mois à 2 ans), 2 mois (2 ans et plus).
if (typeRupture === 'demission'){
if (ancienneteAnnees < 0.5) return { mois: 8/30, texte: '8 jours' };
if (ancienneteAnnees < 2) return { mois: 15/30, texte: '15 jours' };
return { mois: 1, texte: '1 mois' };
}
if (ancienneteAnnees < 0.5) return { mois: 8/30, texte: '8 jours', seuilNonConfirme: true };
if (ancienneteAnnees < 2) return { mois: 1, texte: '1 mois' };
return { mois: 2, texte: '2 mois' };
}
if (convention === 'hcr_maitrise'){
// HCR (IDCC 1979), agents de maîtrise. Mêmes durées pour démission et licenciement : 15 jours (< 6 mois, seuil de licenciement confirmé par une seule source), 1 mois (6 mois à 2 ans), 2 mois (2 ans et plus).
if (ancienneteAnnees < 0.5) return { mois: 15/30, texte: '15 jours', seuilNonConfirme: typeRupture !== 'demission' };
if (ancienneteAnnees < 2) return { mois: 1, texte: '1 mois' };
return { mois: 2, texte: '2 mois' };
}
if (convention === 'hcr_cadre'){
// HCR (IDCC 1979), cadres. Mêmes durées pour démission et licenciement : 1 mois (< 6 mois), 3 mois (6 mois et plus).
return ancienneteAnnees < 0.5 ? { mois: 1, texte: '1 mois' } : { mois: 3, texte: '3 mois' };
}
// --- Syntec (IDCC 1486), ajouté le 14/09/2026 ---
// Article 4.2 de la CCN du 15 décembre 1987, dans sa rédaction issue de l'avenant n°46 du 16 juillet 2021 :
// depuis cet avenant, ETAM et IC relèvent d'un article commun, et le texte vise expressément « en cas de licenciement
// ou de démission » — les durées sont donc réciproques et identiques pour les deux types de rupture (ce n'est pas une
// déduction de notre part). Texte obtenu en verbatim sur Légifrance (conteneur KALICONT000005635173, art. 4.2),
// recoupé sur code.travail.gouv.fr, syntec.fr (fédération), LégiSocial et PayFit — aucune divergence de valeurs.
if (convention === 'syntec_etam'){
// ETAM : moins de 2 ans → 1 mois ; 2 ans et plus → 2 mois.
// Réserve de lecture signalée à l'utilisateur : le texte oppose littéralement « moins de 2 ans » à « plus de 2 ans »,
// laissant le cas des 2 ans pile hors des deux branches. Toutes les sources secondaires consultées lisent
// « 2 ans et plus → 2 mois » (lecture aussi cohérente avec le minimum légal de l'art. L1234-1) : c'est celle retenue ici.
if (ancienneteAnnees < 2){
return { mois: 1, texte: '1 mois', avertissement: 'Les ETAM classés aux coefficients hiérarchiques 400, 450 et 500 relèvent de 2 mois de préavis quelle que soit leur ancienneté : vérifiez votre coefficient sur votre bulletin de paie.' };
}
// L'avertissement sur la lettre du texte n'est affiché qu'au voisinage immédiat de 2 ans, seul cas où la lecture retenue change quelque chose.
if (ancienneteAnnees < 2.1){
return { mois: 2, texte: '2 mois', avertissement: 'Le texte conventionnel vise « plus de 2 ans » sans trancher explicitement le cas d\'une ancienneté de 2 ans pile ; la lecture retenue ici (2 ans et plus → 2 mois) est celle de l\'ensemble des sources consultées.' };
}
return { mois: 2, texte: '2 mois' };
}
if (convention === 'syntec_ic'){
// Ingénieurs et cadres : 3 mois, sans modulation d'ancienneté, en démission comme en licenciement.
return { mois: 3, texte: '3 mois' };
}
// --- Bâtiment (BTP), ajouté le 14/09/2026 ---
// Point de méthode : Légifrance a renvoyé une erreur 403 sur l'intégralité des articles conventionnels du bâtiment
// (art. 7.1 cadres, 8.1 ETAM, 10.11 ouvriers). Les durées ci-dessous reposent donc sur des sources secondaires croisées,
// dont code.travail.gouv.fr (source publique officielle) chaque fois qu'elle expose la donnée, plus LégiSocial,
// Juritravail, Hellowork et convention.fr. Écartée volontairement : la fiche FFB, dont les durées (8 jours en démission
// ouvrier) contredisent toutes les autres sources, y compris code.travail.gouv.fr.
if (convention === 'batiment_cadre'){
// Cadres du bâtiment (IDCC 2420, CCN du 1er juin 2004, art. 7.1) : mêmes durées en démission et en licenciement.
var avertCadreBat = 'La convention des cadres du bâtiment (IDCC 2420) est signalée comme non étendue par code.travail.gouv.fr : elle ne s\'applique de plein droit qu\'aux entreprises adhérentes d\'une organisation signataire. Vérifiez la convention mentionnée sur votre bulletin de paie.';
return ancienneteAnnees < 2
? { mois: 2, texte: '2 mois', avertissement: avertCadreBat }
: { mois: 3, texte: '3 mois', avertissement: avertCadreBat };
}
if (convention === 'batiment_etam'){
// ETAM du bâtiment (IDCC 2609, art. 8.1). Démission : 1 mois (< 2 ans), 2 mois (2 ans et plus) — durées confirmées par
// code.travail.gouv.fr. Licenciement : mêmes durées, plus une tranche de 3 mois réservée aux salariés réunissant
// CUMULATIVEMENT 15 ans d'ancienneté et 55 ans révolus. Cette tranche n'est pas calculée automatiquement (le simulateur
// ne demande pas l'âge) mais signalée en avertissement dès que l'ancienneté atteint 15 ans, sur le même principe que la
// majoration d'âge de la convention Bâtiment cadre sur le simulateur de rupture conventionnelle.
// Point vérifié et volontairement non appliqué : une source (convention.fr) étend les 3 mois à la démission, mais
// code.travail.gouv.fr ne retient que deux tranches en démission — c'est cette lecture officielle qui est suivie ici.
if (ancienneteAnnees < 2){
return { mois: 1, texte: '1 mois' };
}
if (typeRupture !== 'demission' && ancienneteAnnees >= 15){
return { mois: 2, texte: '2 mois', avertissement: 'À partir de 15 ans d\'ancienneté ET 55 ans révolus (les deux conditions doivent être réunies), la convention porte le préavis de licenciement à 3 mois. Le simulateur ne demandant pas votre âge, cette majoration n\'est pas appliquée automatiquement : si vous avez 55 ans ou plus, retenez 3 mois.' };
}
return { mois: 2, texte: '2 mois' };
}
if (convention === 'batiment_ouvrier'){
// Ouvriers du bâtiment, art. 10.11 — IDCC 1596 (jusqu'à 10 salariés) et IDCC 1597 (plus de 10 salariés).
// Vérifié sur les fiches code.travail.gouv.fr des DEUX IDCC : les durées de préavis y sont identiques, la taille de
// l'entreprise ne les modifie pas (elle modifie d'autres droits, pas celui-ci) — d'où une seule option dans le sélecteur.
// Particularité la plus contre-intuitive, et la plus utile à afficher : le préavis de DÉMISSION plafonne à 2 semaines
// quelle que soit l'ancienneté au-delà de 3 mois, là où le préavis de licenciement continue de croître.
if (ancienneteAnnees < 0.25){
return { mois: 2/30, texte: '2 jours' };
}
if (typeRupture === 'demission'){
return { mois: 14/30, texte: '2 semaines', avertissement: 'En démission, le préavis des ouvriers du bâtiment reste de 2 semaines quelle que soit l\'ancienneté au-delà de 3 mois : il n\'augmente pas avec les années, contrairement au préavis de licenciement.' };
}
if (ancienneteAnnees < 0.5){
return { mois: 14/30, texte: '2 semaines' };
}
if (ancienneteAnnees < 2){
return { mois: 1, texte: '1 mois' };
}
return { mois: 2, texte: '2 mois' };
}
return null;
}
function plafondDureeARE(motif, dateRC, age){
// Plafonds de durée d'indemnisation ARE, en mois (sources croisées : France Travail + Unédic, concordantes une fois le coefficient 0,75 réconcilié ; loi n°2026-470 du 11 juin 2026 pour le volet rupture conventionnelle).
// Règle générale (licenciement, fin de CDD, démission légitime, rupture conventionnelle avant le 1er septembre 2026) : 18 mois en-dessous de 55 ans, 22,5 mois de 55 à 56 ans, 27 mois à partir de 57 ans.
// Rupture conventionnelle individuelle prenant effet à partir du 1er septembre 2026 (loi n°2026-470) : 15 mois en-dessous de 55 ans, 20,5 mois à partir de 55 ans (les paliers 55-56 ans et 57 ans et plus sont fusionnés par la réforme, contrairement à la règle générale).
if (motif === 'rc' && dateRC === 'oui'){
return age < 55 ? 15 : 20.5;
}
if (age < 55) return 18;
if (age < 57) return 22.5;
return 27;
}
function dureeAREEstimee(motif, dateRC, age, moisAffiliation){
// Estimation simplifiée de la durée d'indemnisation réelle (pas seulement le plafond) : renvoie { plafond, refMois, mois } où mois est null si l'affiliation n'est pas renseignée.
// Méthode : compte le temps d'affiliation cumulé indiqué par l'utilisateur (plafonné à la période de référence de 24 ou 36 mois selon l'âge), applique le coefficient de 0,75 (Unédic, en vigueur depuis le 1er février 2023), puis applique le plancher garanti de 6 mois et le plafond selon l'âge/le motif.
// Simplification assumée : ne reconstitue pas le détail contrat par contrat ni le plafonnement des périodes non travaillées à 70 % des jours travaillés (Unédic) — le nombre de mois réellement travaillés doit être fourni directement par l'utilisateur.
var refMois = age >= 55 ? 36 : 24;
var plafond = plafondDureeARE(motif, dateRC, age);
if (!moisAffiliation || moisAffiliation <= 0){
return { plafond: plafond, refMois: refMois, mois: null };
}
var moisCap = Math.min(moisAffiliation, refMois);
var moisBruts = moisCap * 0.75;
var moisFinal = Math.max(6, Math.min(moisBruts, plafond));
return { plafond: plafond, refMois: refMois, mois: moisFinal, moisCap: moisCap };
}
function dateLocaleDepuisISO(str){
// 'AAAA-MM-JJ' → Date locale à minuit (null si la date n'existe pas, ex. 31/02).
var m = /^(\d{4})-(\d{2})-(\d{2})$/.exec(str || '');
if (!m) return null;
var d = new Date(+m[1], +m[2] - 1, +m[3]);
if (d.getFullYear() !== +m[1] || d.getMonth() !== +m[2] - 1 || d.getDate() !== +m[3]) return null;
return d;
}
function isoDepuisDateLocale(d){
var pad = function(n){ return String(n).padStart(2, '0'); };
return d.getFullYear() + '-' + pad(d.getMonth() + 1) + '-' + pad(d.getDate());
}
function ajouterMoisCalendaires(d, n){
// De quantième à quantième ; si le jour n'existe pas dans le mois d'arrivée, dernier jour de ce mois (31/08 + 6 mois = 28/02).
var cible = new Date(d.getFullYear(), d.getMonth() + n, 1);
var dernierJour = new Date(cible.getFullYear(), cible.getMonth() + 1, 0).getDate();
cible.setDate(Math.min(d.getDate(), dernierJour));
return cible;
}
function ancienneteDepuisDates(dateEntreeStr, dateSortieStr){
// Corrigé le 19/09/2026 : l'ancien décompte pouvait afficher un nombre de jours négatif (entrée le 31/01, référence le 01/03 → « 1 mois et -2 jours »).
var dEntree = dateLocaleDepuisISO(dateEntreeStr);
var dSortie;
if (dateSortieStr){
dSortie = dateLocaleDepuisISO(dateSortieStr);
} else {
var auj = new Date();
dSortie = new Date(auj.getFullYear(), auj.getMonth(), auj.getDate());
}
if (!dEntree || !dSortie || dSortie < dEntree){
return null;
}
var totalMois = (dSortie.getFullYear() - dEntree.getFullYear()) * 12 + (dSortie.getMonth() - dEntree.getMonth());
while (totalMois > 0 && ajouterMoisCalendaires(dEntree, totalMois) > dSortie){ totalMois -= 1; }
var repere = ajouterMoisCalendaires(dEntree, totalMois);
var jourMs = 1000 * 60 * 60 * 24;
var utc = function(d){ return Date.UTC(d.getFullYear(), d.getMonth(), d.getDate()); };
var years = Math.floor(totalMois / 12);
var months = totalMois % 12;
var days = Math.round((utc(dSortie) - utc(repere)) / jourMs);
return {
years: years,
months: months,
days: days,
totalJours: Math.round((utc(dSortie) - utc(dEntree)) / jourMs),
anneesDecimales: years + months / 12 + days / 365,
dateEntreeISO: isoDepuisDateLocale(dEntree),
dateReferenceISO: isoDepuisDateLocale(dSortie)
};
}
function seuilsAnciennete(dateEntreeISO, dateReferenceISO, listeMois){
// Pour chaque seuil exprimé en mois : date à laquelle il est atteint (entrée + N mois calendaires) et état à la date de référence.
var dEntree = dateLocaleDepuisISO(dateEntreeISO);
var dRef = dateLocaleDepuisISO(dateReferenceISO);
if (!dEntree || !dRef) return [];
return listeMois.map(function(n){
var dSeuil = ajouterMoisCalendaires(dEntree, n);
return { mois: n, dateISO: isoDepuisDateLocale(dSeuil), atteint: dSeuil <= dRef };
});
}
function lireDateSaisie(texte){
// Accepte 05/03/2012, 5/3/2012, 05-03-2012, 05.03.2012, 05032012, 05/03/12, 2012-03-05, 2012/03/05 (et une heure collée à la suite, ignorée).
// Retourne 'AAAA-MM-JJ' ou null.
var t = String(texte || '').trim().split(/[\sT]/)[0];
if (!t) return null;
var j, m, a, r;
if ((r = /^(\d{4})[-\/.](\d{1,2})[-\/.](\d{1,2})$/.exec(t))){ a = +r[1]; m = +r[2]; j = +r[3]; }
else if ((r = /^(\d{1,2})[-\/.](\d{1,2})[-\/.](\d{4}|\d{2})$/.exec(t))){ j = +r[1]; m = +r[2]; a = +r[3]; if (r[3].length === 2){ a += (a <= (new Date().getFullYear() % 100)) ? 2000 : 1900; } }
else if ((r = /^(\d{2})(\d{2})(\d{4})$/.exec(t))){ j = +r[1]; m = +r[2]; a = +r[3]; }
else return null;
if (a < 1900 || a > 2100) return null;
var pad = function(n){ return String(n).padStart(2, '0'); };
var iso = a + '-' + pad(m) + '-' + pad(j);
return dateLocaleDepuisISO(iso) ? iso : null;
}
function dateFRDepuisISO(iso){
var m = /^(\d{4})-(\d{2})-(\d{2})$/.exec(iso || '');
return m ? m[3] + '/' + m[2] + '/' + m[1] : '';
}
function activerSaisieDate(input){
// Insère les barres pendant la frappe (05032012 → 05/03/2012) et normalise une date collée depuis Excel ou un logiciel de paie.
if (!input) return;
input.addEventListener('input', function(e){
if (e && e.inputType && e.inputType.indexOf('delete') === 0) return;
var v = input.value;
if (!/^[\d\/]*$/.test(v)) return;
if (/\/\//.test(v) || v.charAt(0) === '/'){ v = v.replace(/\/{2,}/g, '/').replace(/^\//, ''); input.value = v; }
if (typeof input.selectionStart === 'number' && input.selectionStart !== v.length) return;
var parts = v.split('/');
if (parts.length > 3) return;
var dernier = parts.length - 1;
for (var k = 0; k < parts.length; k++){
var seg = parts[k];
if (k < dernier && seg.length === 1) parts[k] = '0' + seg;
if (k === dernier && seg.length === 1 && ((k === 0 && +seg > 3) || (k === 1 && +seg > 1))) parts[k] = '0' + seg;
if (k < 2 && parts[k].length > 2 && k < dernier) return;
}
var c = parts.join('').slice(0, 8);
var out = c.slice(0, 2);
if (c.length >= 2) out += '/';
out += c.slice(2, 4);
if (c.length >= 4) out += '/';
out += c.slice(4);
input.value = out;
});
input.addEventListener('paste', function(e){
var txt = (e.clipboardData || window.clipboardData);
txt = txt ? txt.getData('text') : '';
var iso = lireDateSaisie(txt);
if (iso){ e.preventDefault(); input.value = dateFRDepuisISO(iso); }
});
input.addEventListener('blur', function(){
var iso = lireDateSaisie(input.value);
if (iso) input.value = dateFRDepuisISO(iso);
});
}

function estimerIndemniteArretMaladie(salaireMensuelBrut, statut, ancienneteAnnees, dureeJours){
if (statut === 'public'){
var carenceJours = BAREME_2026.CARENCE_MALADIE_PUBLIC_JOURS;
var joursPayesPlein = Math.max(0, dureeJours - carenceJours);
var salaireJournalier = salaireMensuelBrut / 30;
var perteCarence = salaireJournalier * Math.min(dureeJours, carenceJours);
return {
statut: statut,
carenceJours: carenceJours,
salaireJournalier: salaireJournalier,
perteCarence: perteCarence,
joursPayesPlein: joursPayesPlein,
totalPercu: salaireJournalier * joursPayesPlein
};
}

// Régime général salarié privé
var salairePlafonne = Math.min(salaireMensuelBrut, BAREME_2026.IJ_MALADIE_PLAFOND_SMIC_MULT * BAREME_2026.SMIC_BRUT_MENSUEL);
var sjb = (salairePlafonne * 3) / BAREME_2026.SJB_DIVISEUR;
var ijBrutJourTheorique = sjb * BAREME_2026.IJ_MALADIE_TAUX;
var ijBrutJour = Math.min(ijBrutJourTheorique, BAREME_2026.IJ_MALADIE_MAX_JOUR);
var ijNetJour = ijBrutJour * (1 - BAREME_2026.IJ_MALADIE_CSG_CRDS_TAUX);

var carenceSecu = BAREME_2026.CARENCE_MALADIE_PRIVE_JOURS;
var joursIndemnisesSecu = Math.max(0, dureeJours - carenceSecu);
var totalIJBrut = ijBrutJour * joursIndemnisesSecu;
var totalIJNet = ijNetJour * joursIndemnisesSecu;

var eligibleMaintien = ancienneteAnnees >= BAREME_2026.MAINTIEN_ANCIENNETE_MIN_ANS;
var palier = eligibleMaintien ? trouverPalierMaintien(Math.floor(ancienneteAnnees)) : null;
var carenceEmployeur = BAREME_2026.MAINTIEN_CARENCE_EMPLOYEUR_JOURS;
var salaireBrutJour = salaireMensuelBrut / 30;
var totalComplementEmployeur = 0;
var joursA90Effectifs = 0, joursA66Effectifs = 0;

if (palier){
var joursApresCarenceEmployeur = Math.max(0, dureeJours - carenceEmployeur);
joursA90Effectifs = Math.min(joursApresCarenceEmployeur, palier.jours);
joursA66Effectifs = Math.min(Math.max(0, joursApresCarenceEmployeur - palier.jours), palier.jours);
var complement90PerJour = Math.max(0, salaireBrutJour * BAREME_2026.MAINTIEN_TAUX_PALIER1 - ijBrutJour);
var complement66PerJour = Math.max(0, salaireBrutJour * BAREME_2026.MAINTIEN_TAUX_PALIER2 - ijBrutJour);
totalComplementEmployeur = complement90PerJour * joursA90Effectifs + complement66PerJour * joursA66Effectifs;
}

return {
statut: 'prive',
salairePlafonne: salairePlafonne,
sjb: sjb,
ijBrutJour: ijBrutJour,
ijNetJour: ijNetJour,
carenceSecu: carenceSecu,
joursIndemnisesSecu: joursIndemnisesSecu,
totalIJBrut: totalIJBrut,
totalIJNet: totalIJNet,
eligibleMaintien: eligibleMaintien,
palier: palier,
carenceEmployeur: carenceEmployeur,
joursA90Effectifs: joursA90Effectifs,
joursA66Effectifs: joursA66Effectifs,
totalComplementEmployeur: totalComplementEmployeur,
totalPercuBrut: totalIJBrut + totalComplementEmployeur,
totalPercuNet: totalIJNet + totalComplementEmployeur
};
}

function estimerJoursRTT(mode, params){
var joursCalendaires = 365;
var weekEnds = 104;
var joursFeries = BAREME_2026.JOURS_FERIES_SEMAINE_2026;
var congesPayes = BAREME_2026.CONGES_PAYES_JOURS_OUVRES;
var joursTravaillablesHorsForfait = joursCalendaires - weekEnds - joursFeries - congesPayes;

if (mode === 'forfait'){
var forfaitAnnuel = params.forfaitAnnuel || BAREME_2026.FORFAIT_JOURS_LEGAL_MAX;
var joursRTT = joursTravaillablesHorsForfait - forfaitAnnuel;
return {
mode: mode,
joursCalendaires: joursCalendaires,
weekEnds: weekEnds,
joursFeries: joursFeries,
congesPayes: congesPayes,
forfaitAnnuel: forfaitAnnuel,
joursRTT: Math.max(0, Math.round(joursRTT * 10) / 10)
};
}

// mode horaire : durée collective hebdomadaire > 35h, réparti sur 5 jours
var dureeHebdo = params.dureeHebdo || 35;
var dureeQuotidienne = dureeHebdo / 5;
var semainesTravaillees = joursTravaillablesHorsForfait / 5;
var heuresRTTAn = Math.max(0, (dureeHebdo - 35) * semainesTravaillees);
var joursRTTHoraire = dureeQuotidienne > 0 ? heuresRTTAn / dureeQuotidienne : 0;
return {
mode: mode,
dureeHebdo: dureeHebdo,
dureeQuotidienne: dureeQuotidienne,
semainesTravaillees: Math.round(semainesTravaillees * 10) / 10,
heuresRTTAn: Math.round(heuresRTTAn * 10) / 10,
joursRTT: Math.round(joursRTTHoraire * 10) / 10
};
}

function pdfCollecterExplication(){
var out = [];
var blocs = document.querySelectorAll('.result .breakdown');
for (var i = 0; i < blocs.length; i++){

if (b.offsetParent === null) continue;
if (b.querySelector('table, a.bridge-link, h2, h3, h4')) continue;

var exclus = b.querySelectorAll('a.bridge-link, button, .pdf-actions, .pdf-privacy-note, .pdf-merci');

texte = texte.replace(/[ \t  ]+/g, ' ').replace(/\s*\n\s*/g, '\n').trim();
if (texte.length >= 15) out.push(texte);
}

}
function pdfCollecterSources(){
var out = [], vus = {};
var badges = document.querySelectorAll('.source-badge');
for (var i = 0; i < badges.length; i++){
var a = badges[i].previousElementSibling;
if (!a || a.tagName !== 'A') continue;
var href = a.getAttribute('href') || '';
if (!/^https?:\/\//.test(href) || vus[href]) continue;
vus[href] = true;
var label = (a.textContent || '').replace(/\s+/g, ' ').trim();
if (label.indexOf(' ') === -1 && label.indexOf('.') !== -1 && a.parentNode){
var contexte = a.parentNode.querySelector('strong');
if (contexte && contexte.textContent.trim()){ label = contexte.textContent.replace(/\s*:\s*$/, '').trim() + ' — ' + label; }
}
label = label.replace(/^(l'|la |le |les |du |de la |des |une |un )/i, '');
label = label.charAt(0).toUpperCase() + label.slice(1);
out.push({ label: label, url: href, rang: pdfRangSource(href), ordre: out.length });
}
out.sort(function(a, b){ return a.rang - b.rang || a.ordre - b.ordre; });
return out.slice(0, 8);
}
function pdfRangSource(url){
if (/legifrance\.gouv\.fr/.test(url)) return 0;
if (/code\.travail\.gouv\.fr/.test(url)) return 1;
if (/service-public\.(gouv\.)?fr/.test(url)) return 2;
if (/unedic\.org|francetravail\.fr|urssaf\.fr|ameli\.fr|\.gouv\.fr/.test(url)) return 3;
return 4;
}

function estimerPartParticipation(montantGlobal, cle, salaireBrutAnnuel, masseSalarialeTotale, effectifTotal, moisPresence){
// Réserve spéciale de participation (RSP) — répartition individuelle, PAS le calcul du montant global
// (celui-ci dépend de données de bilan — bénéfice net, capitaux propres, valeur ajoutée — non disponibles
// pour un salarié individuel, cf. contenu explicatif de la page).
// Sources croisées (art. L3324-5 et D3324-10 du Code du travail, via LégiFiscal/Previssima/Éres Group/culture-rh.com,
// concordantes) : plafond de salaire pris en compte pour la répartition proportionnelle = 3×PASS (post loi PACTE 2019,
// remplace l'ancien plafond de 4×PASS antérieur à 2019) ; plafond individuel de versement = 75 % du PASS.
var quotitePresence = Math.min(Math.max((moisPresence || 12) / 12, 0), 1);
var plafondSalairePriseEnCompte = 3 * BAREME_2026.PASS_ANNUEL;
var plafondIndividuel = 0.75 * BAREME_2026.PASS_ANNUEL;
var salairePlafonne = salaireBrutAnnuel > 0 ? Math.min(salaireBrutAnnuel, plafondSalairePriseEnCompte) : 0;
var partUniforme = (effectifTotal > 0) ? (montantGlobal / effectifTotal) * quotitePresence : null;
var partSalaire = (masseSalarialeTotale > 0 && salaireBrutAnnuel > 0) ? montantGlobal * (salairePlafonne * quotitePresence) / masseSalarialeTotale : null;
var part = null;
if (cle === 'uniforme'){
part = partUniforme;
} else if (cle === 'salaire'){
part = partSalaire;
} else if (cle === 'mixte' && partUniforme !== null && partSalaire !== null){
part = 0.5 * partUniforme + 0.5 * partSalaire;
}
var donneesInsuffisantes = (part === null);
var partAvantPlafond = part;
var ecretee = false;
if (part !== null && part > plafondIndividuel){
part = plafondIndividuel;
ecretee = true;
}
return {
part: part,
partAvantPlafond: partAvantPlafond,
ecretee: ecretee,
plafondIndividuel: plafondIndividuel,
plafondSalairePriseEnCompte: plafondSalairePriseEnCompte,
salairePlafonne: salairePlafonne,
quotitePresence: quotitePresence,
partUniforme: partUniforme,
partSalaire: partSalaire,
donneesInsuffisantes: donneesInsuffisantes
};
}

// Délais pour saisir le conseil de prud'hommes (saisir-les-prudhommes.html).
// Sources : L1471-1, L1237-14, L3245-1, L1234-20, L1134-5 du Code du travail ; art. 2224 et 2226 du Code civil ; service-public.gouv.fr F2360.
var DELAIS_PRUDHOMMES = {
rupture:{mois:12, depart:"de la notification de la rupture (date de première présentation de la lettre de licenciement, par exemple)", article:"article L1471-1 du Code du travail"},
rc:{mois:12, depart:"de la date d'homologation de la rupture conventionnelle", article:"article L1237-14 du Code du travail"},

salaires:{mois:36, depart:"du jour où vous avez connu (ou auriez dû connaître) les faits, en pratique la date à laquelle la somme aurait dû être payée", article:"article L3245-1 du Code du travail"},
stc:{mois:6, depart:"de la signature du reçu pour solde de tout compte", article:"article L1234-20 du Code du travail"},

harcelement:{mois:60, depart:"de la date du dernier fait de harcèlement", article:"article 2224 du Code civil"},
corporel:{mois:120, depart:"de la date de consolidation du dommage", article:"article 2226 du Code civil"}
};
function ajouterMoisDate(date, mois){
var y = date.getFullYear(), m = date.getMonth() + mois, d = date.getDate();
var dernierJour = new Date(y, m + 1, 0).getDate();
return new Date(y, m, Math.min(d, dernierJour));
}
function delaiPrudhommes(type, dateDepartISO, aujourdhui){
var regle = DELAIS_PRUDHOMMES[type];
if (!regle || !/^\d{4}-\d{2}-\d{2}$/.test(dateDepartISO || '')) return null;
var parts = dateDepartISO.split('-');
var depart = new Date(+parts[0], +parts[1] - 1, +parts[2]);
if (isNaN(depart.getTime())) return null;
var ref = aujourdhui ? new Date(aujourdhui.getFullYear(), aujourdhui.getMonth(), aujourdhui.getDate()) : new Date(new Date().setHours(0,0,0,0));
var limite = ajouterMoisDate(depart, regle.mois);
var joursRestants = Math.round((limite - ref) / 86400000);
return {mois:regle.mois, depart:depart, limite:limite, joursRestants:joursRestants, futur:depart > ref, regle:regle};
}

// ISRC — indemnité spécifique de rupture conventionnelle dans la fonction publique (rupture-conventionnelle-fonction-publique.html).
// Décret n° 2019-1596 du 31/12/2019 modifié par le décret n° 2026-746 du 6 août 2026 (en vigueur le 8 août 2026).
// Sources : fonction-publique.gouv.fr (fiche « Dispositif de rupture conventionnelle », maj 02/09/2026), service-public.gouv.fr F31094 (vérifiée le 08/08/2026).
// Plancher par année complète d'ancienneté, par tranches (fraction de la rémunération mensuelle brute = rémunération brute annuelle N-1 / 12) ; plafond par année, dans la limite de 24 ans.
var BAREME_ISRC = {
actuel:{libelle:"barème en vigueur depuis le 8 août 2026", tranches:[{jusqua:10, fraction:1/6, texte:"1/6"},{jusqua:15, fraction:1/5, texte:"1/5"},{jusqua:20, fraction:1/4, texte:"1/4"},{jusqua:24, fraction:1/3, texte:"1/3"}], plafondDiviseur:24},
ancien:{libelle:"ancien barème (décret de 2019, avant le 8 août 2026)", tranches:[{jusqua:10, fraction:1/4, texte:"1/4"},{jusqua:15, fraction:2/5, texte:"2/5"},{jusqua:20, fraction:1/2, texte:"1/2"},{jusqua:24, fraction:3/5, texte:"3/5"}], plafondDiviseur:12}
};
var ISRC_ANCIENNETE_MAX = 24;
function isrcPlancherPlafond(remunerationBruteAnnuelle, anneesCompletes, cleBareme){
var bareme = BAREME_ISRC[cleBareme || 'actuel'];
var rem = Number(remunerationBruteAnnuelle);
var annees = Math.floor(Number(anneesCompletes));
if (!bareme || !(rem > 0) || !(annees >= 1)) return null;
var retenues = Math.min(annees, ISRC_ANCIENNETE_MAX);
var mensuel = rem / 12;
var plancher = 0, detail = [], debut = 0;
bareme.tranches.forEach(function(t){
var n = Math.max(0, Math.min(retenues, t.jusqua) - debut);
if (n > 0){
var montant = n * t.fraction * mensuel;
plancher += montant;
detail.push({de:debut + 1, a:debut + n, annees:n, fractionTexte:t.texte, montant:montant});
}
debut = t.jusqua;
});
var plafond = rem / bareme.plafondDiviseur * retenues;
return {plancher:plancher, plafond:plafond, mensuel:mensuel, anneesRetenues:retenues, anneesSaisies:annees, ecrete:annees > ISRC_ANCIENNETE_MAX, detail:detail, libelle:bareme.libelle};
}

// Différé spécifique d'indemnisation ARE (règlement d'assurance chômage, art. 21 § 1).
// Diviseur 2026 = 111,8 (circulaire Unédic n° 2026-01, revalorisé chaque année avec le plafond de la sécurité sociale).
// Nombre entier de jours calendaires ; plafond 150 jours, 75 jours en cas de rupture pour motif économique.
var DIFFERE_ARE_2026 = {DIVISEUR:111.8, PLAFOND_JOURS:150, PLAFOND_JOURS_ECO:75, DELAI_ATTENTE_JOURS:7, PLAFOND_DIFFERE_CP_JOURS:30};
function differeSpecifiqueARE(montantSupraLegal, motifEconomique){
var montant = (isFinite(montantSupraLegal) && montantSupraLegal > 0) ? montantSupraLegal : 0;
var plafond = motifEconomique ? DIFFERE_ARE_2026.PLAFOND_JOURS_ECO : DIFFERE_ARE_2026.PLAFOND_JOURS;
var joursBruts = Math.floor(montant / DIFFERE_ARE_2026.DIVISEUR + 1e-9);
var jours = Math.min(joursBruts, plafond);
return {montant:montant, diviseur:DIFFERE_ARE_2026.DIVISEUR, joursBruts:joursBruts, jours:jours, plafond:plafond, plafonne:joursBruts > plafond, montantAuPlafond:plafond * DIFFERE_ARE_2026.DIVISEUR};
}

function normalizeSearchStr(s){
return (s || '').toLowerCase().normalize('NFD').replace(/[\u0300-\u036f]/g, '');
}

function wordSetOf(s){
return normalizeSearchStr(s).split(/[^a-z0-9]+/).filter(Boolean);
}

function searchPages(query){
var q = normalizeSearchStr(query).trim();
if (!q) return [];
var words = q.split(/\s+/).filter(Boolean);
var currentPath = window.location.pathname.split('/').pop() || 'index.html';
var results = [];
SEARCH_INDEX.forEach(function(entry){
if (entry.url === currentPath) return;
var hayTitle = normalizeSearchStr(entry.title);
var hayAll = hayTitle + ' ' + normalizeSearchStr(entry.kw) + ' ' + normalizeSearchStr(entry.cat);
var titleWords = wordSetOf(entry.title);
var allWords = titleWords.concat(wordSetOf(entry.kw), wordSetOf(entry.cat));
var score = 0;
var matchAll = words.every(function(w){
if (w.length <= 3){
if (titleWords.indexOf(w) !== -1) { score += 10; return true; }
if (allWords.indexOf(w) !== -1) { score += 1; return true; }
return false;
}
if (hayTitle.indexOf(w) !== -1) { score += 10; return true; }
if (hayAll.indexOf(w) !== -1) { score += 1; return true; }
return false;
});
if (matchAll) results.push({entry: entry, score: score});
});
results.sort(function(a, b){ return b.score - a.score; });
return results.map(function(r){ return r.entry; }).slice(0, 8);
}

if (!list || !empty) return;
list.innerHTML = '';
if (!query) { empty.hidden = true; return; }
if (results.length === 0) { empty.hidden = false; return; }
empty.hidden = true;
results.forEach(function(entry, i){
var li = document.createElement('li');
var a = document.createElement('a');
a.href = entry.url;
if (i === 0) a.className = 'active';
var titleSpan = document.createElement('span');
titleSpan.className = 'sr-title';
titleSpan.textContent = entry.title;
var catSpan = document.createElement('span');
catSpan.className = 'sr-cat';
catSpan.textContent = entry.cat;
a.appendChild(titleSpan);
a.appendChild(catSpan);
li.appendChild(a);
list.appendChild(li);
});
}

function initSiteSearch(){
var toggle = document.getElementById('search-toggle');
var overlay = document.getElementById('search-overlay');
var closeBtn = document.getElementById('search-close');
var input = document.getElementById('search-input');
if (!toggle || !overlay || !input) return;

function isTypingElsewhere(){
var t = document.activeElement;
return t && ['INPUT', 'TEXTAREA', 'SELECT'].indexOf(t.tagName) !== -1 && t !== input;
}

function openSearch(){
overlay.hidden = false;
document.body.style.overflow = 'hidden';
setTimeout(function(){ input.focus(); }, 10);
}
function closeSearch(){
overlay.hidden = true;
document.body.style.overflow = '';
input.value = '';
renderSearchResults([], '');
}

toggle.addEventListener('click', function(){
if (overlay.hidden) { openSearch(); } else { closeSearch(); }
});
closeBtn.addEventListener('click', closeSearch);
overlay.addEventListener('click', function(e){
if (e.target === overlay) closeSearch();
});
document.addEventListener('keydown', function(e){
if (e.key === 'Escape' && !overlay.hidden) closeSearch();
if (e.key === '/' && overlay.hidden && !isTypingElsewhere()) {
e.preventDefault();
openSearch();
}
});
input.addEventListener('input', function(){
var q = input.value;
renderSearchResults(searchPages(q), q);
});
input.addEventListener('keydown', function(e){
if (e.key === 'Enter') {
var first = document.querySelector('#search-results a');
if (first) { window.location.href = first.getAttribute('href'); }
}
});
}
document.addEventListener('DOMContentLoaded', initSiteSearch);
/* Défilement vers le résultat + touche Entrée (19/09/2026) — uniquement après une action volontaire
   (clic sur « Calculer » ou Entrée dans un champ), jamais au recalcul automatique ni au pré-remplissage. */
function defilerVersResultat(card){
if (!card) return;
var res = card.querySelector('.result.show');
if (!res) return;
var r = res.getBoundingClientRect();
var header = document.querySelector('header.site');
var haut = (header ? header.getBoundingClientRect().height : 0) + 24;
var vh = window.innerHeight || document.documentElement.clientHeight;
if (r.top >= haut && r.bottom <= vh) return;
var doux = !(window.matchMedia && window.matchMedia('(prefers-reduced-motion: reduce)').matches);
window.scrollTo({ top: Math.max(0, window.pageYOffset + r.top - haut), behavior: doux ? 'smooth' : 'auto' });
}
function initResultatAccessible(){
document.querySelectorAll('.card .result').forEach(function(res){
if (!res.hasAttribute('aria-live')) res.setAttribute('aria-live', 'polite');
});
document.addEventListener('click', function(e){
var btn = e.target.closest ? e.target.closest('button.compute') : null;
if (!btn || btn.classList.contains('pdf')) return;
var card = btn.closest('.card');
window.requestAnimationFrame(function(){ defilerVersResultat(card); });
});
document.addEventListener('keydown', function(e){
if (e.key !== 'Enter' || e.isComposing || e.shiftKey || e.ctrlKey || e.altKey || e.metaKey) return;
var el = e.target;
if (!el || el.tagName !== 'INPUT') return;
var type = (el.getAttribute('type') || 'text').toLowerCase();
if (['checkbox','radio','button','submit','file','range'].indexOf(type) !== -1) return;
var card = el.closest('.card');
if (!card) return;
var btn = card.querySelector('button.compute:not(.pdf)');
if (!btn || btn.disabled) return;
e.preventDefault();
btn.click();
});
}
document.addEventListener('DOMContentLoaded', initResultatAccessible);
